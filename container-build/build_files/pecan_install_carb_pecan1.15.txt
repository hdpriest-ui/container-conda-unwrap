# Option A hybrid build: conda base + renv R package layer
#
# WHAT THIS PRODUCES (in /mnt/packed/):
#   pecan-all.tar.gz  - conda base env (R runtime + system libs) with three extras embedded:
#                         ${env}/renv.lock              - exact R package versions
#                         ${env}/renv-source-cache/     - compiled package cache
#                         ${env}/renv-source-cache/sources/ - source tarballs for recompilation
#
# DESTINATION SETUP: see CARB-Demo-PEcAn-env-walkthrough-hybrid.md
#
# v1.15 changes from v1.14:
#   - conda install now also pulls fftw, libtiff, and xorg-libx11. imager's
#     SystemRequirements are fftw3, libtiff, and X11; PKG_CONFIG_PATH already
#     points at ${env}/lib/pkgconfig (exported below), so imager's configure
#     script picks these up via pkg-config with no extra configure.vars.
#   - conda install now also pulls libgdal-hdf5 and libgdal-netcdf, the GDAL
#     raster driver plugins for HDF5 and netCDF, so gdal-backed reads/writes
#     of those formats work without falling back to a driver-less build.
#   - renv::install now also installs doMC, matrixStats, RcppRoll, imager,
#     and WGCNA. WGCNA Imports two Bioconductor-only packages (impute,
#     preprocessCore) that don't exist on CRAN/pecan r-universe/RSPM, so
#     those are installed first via renv's bioc:: remote syntax (using the
#     BiocManager already renv-installed in the CRAN-utility batch below)
#     before WGCNA itself is requested.
#
# v1.14 changes from v1.13:
#   - conda install now also pulls zeromq, numpy, pandas, shapely, and tqdm.
#     zeromq is the system ZeroMQ (libzmq) that clustermq links against for
#     HPC/cluster job dispatch (optional per clustermq's SystemRequirements,
#     but bundling libzmq from source inside the R build is slower and less
#     reliable than linking the conda-provided one); numpy/pandas/shapely/tqdm
#     are added explicitly even though geopandas/fiona/dask already pull most
#     of them in transitively, so they're pinned as first-class requirements
#     rather than incidental.
#   - renv::install now also installs clustermq, config, crew, crew.cluster,
#     tarchetypes, and targets — the targets-based pipeline/orchestration
#     stack, plus crew/crew.cluster and clustermq for distributed execution
#     on HPC schedulers.
#
# v1.13 changes from v1.12:
#   - conda install now also pulls libsecret, glib, libboost-devel, and
#     pkg-config. renv::restore() at the destination previously relied on
#     these being present as ambient system libraries (true on our own
#     hosts, false on at least one third-party HPC). keyring needs libsecret
#     to link, and specifically needs glib (not just the libglib runtime lib
#     that libsecret pulls in transitively) for glibconfig.h; arrow's
#     bundled C++ build needs boost headers/libs when it can't find a
#     prebuilt libarrow; pkg-config itself must not depend on whatever the
#     destination host happens to have on its system PATH.

# r-universe (pecanproject.r-universe.dev) stamps built packages' DESCRIPTION with RemoteType/RemoteUrl/RemoteSha/RemoteSubdir provenance from the pecan monorepo, even though they install as ordinary Source: "Repository" packages. renv::restore() sees those fields and misidentifies the record as needing git-style subdir extraction, which fails offline (the cached source tarball is version-named and already flattened, not SHA-named/nested). Strip the stray fields here so the shipped lockfile restores via the normal repository path everywhere it's used.
Rscript -e "lockfile <- renv:::renv_lockfile_read('${CONDA_ENV_NAME}/renv.lock'); fields <- c('RemoteType', 'RemoteUrl', 'RemoteHost', 'RemoteRepo', 'RemoteUsername', 'RemoteRef', 'RemoteSha', 'RemoteSubdir', 'RemotePkgRef'); for (name in names(lockfile\$Packages)) { rec <- lockfile\$Packages[[name]]; if (identical(rec\$Source, 'Repository')) { rec[fields] <- NULL; lockfile\$Packages[[name]] <- rec } }; renv:::renv_lockfile_write(lockfile, file = '${CONDA_ENV_NAME}/renv.lock')"
